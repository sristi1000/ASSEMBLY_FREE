singularity exec --bind /home/jagat/TOOLS/humann_dbs:/dbs /home/jagat/TOOLS/humann3.sif humann_renorm_table --input 03_Humann3/Chemical_03_humann/Chemical_03_merged.fastp_genefamilies.tsv --units cpm --output 03_Humann3/Chemical_03_humann/Chemical_03_genefamilies_cpm.tsv

# 3 Regrouped to KO
singularity exec   --bind /home/jagat/TOOLS/humann_dbs:/dbs   /home/jagat/TOOLS/humann3.sif   humann_regroup_table     --input 03_Humann3/Chemical_03_humann/Chemical_03_genefamilies_cpm.tsv     --custom /dbs/utility_mapping/map_ko_uniref90.txt.gz     --output 03_Humann3/Chemical_03_humann/Chemical_03_ko_cpm.tsv

# 4 Attach readable name
singularity exec   --bind /home/jagat/TOOLS/humann_dbs:/dbs   /home/jagat/TOOLS/humann3.sif   humann_rename_table     --input 03_Humann3/Chemical_03_humann/Chemical_03_ko_cpm.tsv     --custom /dbs/utility_mapping/map_ko_name.txt.gz     --output 03_Humann3/Chemical_03_humann/Chemical_03_ko_cpm_named.tsv

# 5 Regroup COG
singularity exec   --bind /home/jagat/TOOLS/humann_dbs:/dbs   /home/jagat/TOOLS/humann3.sif   humann_regroup_table     --input 03_Humann3/Chemical_03_humann/Chemical_03_genefamilies_cpm.tsv     --custom /dbs/utility_mapping/map_eggnog_uniref90.txt.gz     --output 03_Humann3/Chemical_03_humann/Chemical_03_cog_cpm.tsv

# 6 Attach readable name
singularity exec   --bind /home/jagat/TOOLS/humann_dbs:/dbs   /home/jagat/TOOLS/humann3.sif   humann_rename_table     --input 03_Humann3/Chemical_03_humann/Chemical_03_cog_cpm.tsv     --custom /dbs/utility_mapping/map_eggnog_name.txt.gz     --output 03_Humann3/Chemical_03_humann/Chemical_03_cog_cpm_named.tsv

# 7 Normalized pathway abundance and relative abundance
singularity exec   --bind /home/jagat/TOOLS/humann_dbs:/dbs   /home/jagat/TOOLS/humann3.sif   humann_renorm_table     --input 03_Humann3/Chemical_03_humann/Chemical_03_merged.fastp_pathabundance.tsv     --units relab     --output 03_Humann3/Chemical_03_humann/Chemical_03_pathabundance_relab.tsv

# 8 Clear pathway defination or name
singularity exec   --bind /home/jagat/TOOLS/humann_dbs:/dbs   /home/jagat/TOOLS/humann3.sif   humann_rename_table     --input 03_Humann3/Chemical_03_humann/Chemical_03_pathabundance_relab.tsv     --names metacyc-pwy     --output 03_Humann3/Chemical_03_humann/Chemical_03_pathabundance_named.tsv
